# The tests of PHP's suite that the lab's test services/php leaves out,
# each group with the reason it fails on Dukovo. A line is a test file's
# path from the top of the suite (`ext/standard/tests/file/fgets.phpt`), or
# a folder's, which ends in a slash (`ext/sockets/tests/`): the lab test
# runs the tests of the suite whose paths start with none of them. When
# a gap closes, its lines go: the gaps waiting on work are in
# docs/deferred.md.
#
# The list is made from runs of the whole suite under emulation (ADR
# 0499). Each test that failed there, or ran for longer than its time, was
# read for its reason before its line was added; a test that fails only
# because emulation is slow, as one that times itself does, has no line.
# The lab test runs the suite on hardware, and a test that fails there gets
# a line once its reason is known.

# mlibc's getprotobyname(3) and getprotobynumber(3) find no protocol, and
# its getservbyport(3) ignores the protocol it is given (it names "http"
# for UDP port 80 as well).
ext/standard/tests/network/getprotobyname_basic.phpt
ext/standard/tests/network/getprotobynumber_basic.phpt
ext/standard/tests/general_functions/getservbyport_variation1.phpt

# The C library has no res_search(3), so the build has no dns_get_record(),
# dns_check_record() or dns_get_mx(), which these tests call (deferred.md).
ext/standard/tests/network/ghsa-www2-q4fc-65wf.phpt
ext/standard/tests/network/dns_check_record_error_conditions.phpt
ext/standard/tests/network/dns_get_record_error_conditions.phpt

# Socket options the kernel does not offer answer ENOPROTOOPT (92):
# TCP_CONGESTION, SO_ZEROCOPY and IPV6_PKTINFO and IPV6_RECVPKTINFO; the
# test of an option that must be refused for a different reason expects
# EACCES. gh21161 also finds that the C library's getaddrinfo(3) takes an
# empty name, which glibc's refuses, so PHP does not warn "Host lookup
# failed" before the option is refused.
ext/sockets/tests/socket_tcp_congestion.phpt
ext/sockets/tests/socket_sendto_zerocopy.phpt
ext/sockets/tests/socket_set_option_in6_pktinfo.phpt
ext/sockets/tests/gh21161.phpt
ext/sockets/tests/socket_sendrecvmsg_multi_msg.phpt
ext/sockets/tests/socket_sendrecvmsg_multi_msg-unix.phpt
ext/sockets/tests/socket_set_option_error_socket_option.phpt

# The ioctl that gives the address of a network interface by its number
# (the multicast options' interface argument) answers ENOTTY (25).
ext/sockets/tests/socket_getopt.phpt
ext/sockets/tests/mcast_ipv4_send.phpt
ext/sockets/tests/mcast_ipv4_send_error.phpt

# A datagram the machine sends to an interface-local IPv6 multicast group
# (ff01::114) that its own socket joined never reaches that socket: the
# receive waits until its time runs out.
ext/sockets/tests/mcast_ipv6_recv_limited.phpt

# The kernel takes what Linux refuses: sendmsg(2) with flags it has no
# meaning for succeeds, and socketpair(2) for AF_INET answers "Address
# family not supported by protocol" where PHP's test expects the C library's
# "not supported" message.
ext/sockets/tests/socket_sendrecvmsg_error.phpt
ext/sockets/tests/socket_create_pair-wrongparams.phpt

# mlibc's wait3() is not implemented (it says so on the standard error
# output), which pcntl_wait() and pcntl_waitpid() with a resource usage

ext/pcntl/tests/002.phpt
ext/pcntl/tests/bug72154.phpt
ext/pcntl/tests/pcntl_wait.phpt
ext/pcntl/tests/pcntl_wait_rusage1.phpt

# mlibc's sysconf() has no number for some names posix_sysconf() asks
# (it answers -1 and says so).
ext/posix/tests/posix_sysconf.phpt
